CREATE TABLE asset_discoveries ( asset_id TEXT NOT NULL REFERENCES assets(id), operation_id TEXT NOT NULL, caller_scope TEXT NOT NULL, intent_id TEXT NOT NULL REFERENCES intents(id), attempt_id TEXT NOT NULL REFERENCES attempts(id), discovered_as TEXT NOT NULL, note TEXT NOT NULL, discovered_at INTEGER NOT NULL, PRIMARY KEY (asset_id, operation_id) )
## table asset_imports (on asset_imports)
CREATE TABLE asset_imports ( asset_id TEXT NOT NULL REFERENCES assets(id), operation_id TEXT NOT NULL, caller_scope TEXT NOT NULL, source_url TEXT NOT NULL, imported_as TEXT NOT NULL, note TEXT NOT NULL, imported_at INTEGER NOT NULL, PRIMARY KEY (asset_id, operation_id), CHECK (length(source_url) > 0) )
## table asset_scope_labels (on asset_scope_labels)
CREATE TABLE asset_scope_labels ( asset_id TEXT PRIMARY KEY REFERENCES assets(id), state TEXT NOT NULL, reason TEXT, set_by TEXT NOT NULL, set_at INTEGER NOT NULL, CHECK (state IN ('in', 'out')) )
## table assets (on assets)
CREATE TABLE assets ( id TEXT PRIMARY KEY, target_id TEXT NOT NULL, kind TEXT NOT NULL, identity TEXT NOT NULL, host TEXT NOT NULL, port INTEGER NOT NULL, scheme TEXT, method TEXT, path TEXT, first_seen_at INTEGER NOT NULL, UNIQUE (target_id, kind, identity), CHECK ( (kind = 'host' AND scheme IS NULL AND method IS NULL AND path IS NULL) OR (kind = 'endpoint' AND scheme IS NOT NULL AND method IS NOT NULL AND path IS NOT NULL) ) )
## table attempt_recoveries (on attempt_recoveries)
CREATE TABLE attempt_recoveries ( attempt_id TEXT NOT NULL REFERENCES attempts(id), observed_status TEXT NOT NULL, prior_status TEXT NOT NULL, resulting_status TEXT NOT NULL, disposition TEXT NOT NULL, detail TEXT NOT NULL, harness_session_id TEXT, fence_before INTEGER NOT NULL, fence_after INTEGER NOT NULL, observations_at_recovery INTEGER NOT NULL, first_reconciled_at INTEGER NOT NULL, last_reconciled_at INTEGER NOT NULL, times INTEGER NOT NULL, PRIMARY KEY (attempt_id, observed_status, prior_status, resulting_status, disposition, fence_before, fence_after) )
## table attempts (on attempts)
CREATE TABLE attempts ( id TEXT PRIMARY KEY, intent_id TEXT NOT NULL REFERENCES intents(id), target_id TEXT NOT NULL, worker_id TEXT NOT NULL, harness_session_id TEXT, provider_id TEXT NOT NULL, model_id TEXT NOT NULL, status TEXT NOT NULL, started_at INTEGER NOT NULL, ended_at INTEGER, write_fence INTEGER NOT NULL DEFAULT 0, recovery_disposition TEXT, recovered_at INTEGER, replaces_attempt_id TEXT REFERENCES attempts(id), harness_instance_id TEXT, model_options TEXT )
## table command_receipts (on command_receipts)
CREATE TABLE command_receipts ( key TEXT PRIMARY KEY, caller_scope TEXT NOT NULL, operation TEXT NOT NULL, operation_id TEXT NOT NULL, request_hash TEXT NOT NULL, result TEXT NOT NULL, created_at INTEGER NOT NULL )

## table goal_sessions (on goal_sessions)

## table goals (on goals)
CREATE TABLE goals ( id TEXT PRIMARY KEY, target_id TEXT NOT NULL, description TEXT NOT NULL, created_at INTEGER NOT NULL, name TEXT NOT NULL DEFAULT '', status TEXT NOT NULL DEFAULT 'active', ended_at INTEGER, ended_by TEXT, end_reason TEXT )
## table host_scope_labels (on host_scope_labels)
CREATE TABLE host_scope_labels ( target_id TEXT NOT NULL, host TEXT NOT NULL, state TEXT NOT NULL, reason TEXT, set_by TEXT NOT NULL, set_at INTEGER NOT NULL, PRIMARY KEY (target_id, host), CHECK (state IN ('in', 'out')), CHECK (length(host) > 0) )
## table import_sessions (on import_sessions)
CREATE TABLE import_sessions ( import_id TEXT PRIMARY KEY, target_id TEXT NOT NULL, harness_session_id TEXT NOT NULL, harness_instance_id TEXT, model TEXT NOT NULL, caller_scope TEXT NOT NULL, opened_at INTEGER NOT NULL, session_created_at INTEGER, ended_at INTEGER, ended_reason TEXT CHECK (ended_reason IN ('operator', 'shutdown', 'refused', 'open_failed')), model_options TEXT, pages TEXT, agent_id TEXT NOT NULL DEFAULT 'importer' CHECK (length(agent_id) > 0), CHECK (length(harness_session_id) > 0) )
## table intent_anchors (on intent_anchors)
CREATE TABLE intent_anchors ( intent_id TEXT NOT NULL REFERENCES intents(id), observation_id TEXT NOT NULL REFERENCES observations(id), position INTEGER NOT NULL, created_at INTEGER NOT NULL, PRIMARY KEY (intent_id, observation_id), CHECK (position >= 0) )
## table intents (on intents)
CREATE TABLE intents ( id TEXT PRIMARY KEY, goal_id TEXT NOT NULL REFERENCES goals(id), target_id TEXT NOT NULL, question TEXT NOT NULL, created_at INTEGER NOT NULL, addresses_dimension TEXT REFERENCES observation_dimensions(id), status TEXT NOT NULL DEFAULT 'open', claimed_by_attempt_id TEXT, suggested_instance_id TEXT, suggested_model TEXT, suggested_model_options TEXT )
## table observation_dimension_links (on observation_dimension_links)
CREATE TABLE observation_dimension_links ( observation_id TEXT NOT NULL REFERENCES observations(id), dimension_id TEXT NOT NULL REFERENCES observation_dimensions(id), address TEXT NOT NULL, created_at INTEGER NOT NULL, PRIMARY KEY (observation_id, dimension_id) )
## table observation_dimensions (on observation_dimensions)
CREATE TABLE observation_dimensions ( id TEXT PRIMARY KEY, observation_id TEXT NOT NULL REFERENCES observations(id), position INTEGER NOT NULL, text TEXT NOT NULL, created_at INTEGER NOT NULL, UNIQUE (observation_id, position), CHECK (position >= 0) )
## table observations (on observations)
CREATE TABLE observations ( id TEXT PRIMARY KEY, sequence INTEGER NOT NULL, record_format TEXT NOT NULL, target_id TEXT NOT NULL, intent_id TEXT NOT NULL REFERENCES intents(id), attempt_id TEXT NOT NULL REFERENCES attempts(id), harness_session_id TEXT NOT NULL, worker_id TEXT NOT NULL, tested_claim TEXT NOT NULL, outcome TEXT NOT NULL, interpretation TEXT NOT NULL, limitations TEXT NOT NULL, untested_dimensions TEXT NOT NULL, recorded_at INTEGER NOT NULL, setup TEXT, cases TEXT, summary_conditions TEXT, summary_observed_behavior TEXT, summary_evidence_refs TEXT, CHECK ( (record_format = 'case_based' AND setup IS NOT NULL AND cases IS NOT NULL AND summary_conditions IS NULL AND summary_observed_behavior IS NULL AND summary_evidence_refs IS NULL) OR (record_format = 'summary' AND setup IS NULL AND cases IS NULL AND summary_conditions IS NOT NULL AND summary_observed_behavior IS NOT NULL AND summary_evidence_refs IS NOT NULL) ) )
## table outbox (on outbox)
CREATE TABLE outbox ( id TEXT PRIMARY KEY, sequence INTEGER NOT NULL, target_id TEXT NOT NULL, request_type TEXT NOT NULL, payload TEXT NOT NULL, recovery_policy TEXT NOT NULL, caused_by_event_id TEXT NOT NULL REFERENCES events(id), status TEXT NOT NULL, attempts INTEGER NOT NULL, lease_owner TEXT, lease_expires_at INTEGER, next_attempt_at INTEGER, last_error TEXT, enqueued_at INTEGER NOT NULL, completed_at INTEGER, CHECK (status IN ('pending', 'in_flight', 'succeeded', 'failed', 'cancelled')), CHECK (recovery_policy IN ('safe_to_retry', 'idempotency_key', 'reconcile_first', 'do_not_repeat')), CHECK (attempts >= 0) )
## table planner_lineages (on planner_lineages)

## table scope_entries (on scope_entries)
CREATE TABLE scope_entries ( id TEXT PRIMARY KEY, target_id TEXT NOT NULL, effect TEXT NOT NULL, kind TEXT NOT NULL, value TEXT NOT NULL, source TEXT NOT NULL, note TEXT, created_at INTEGER NOT NULL, UNIQUE (target_id, effect, kind, value), CHECK (effect IN ('in', 'out')), CHECK (kind IN ('subtree', 'host', 'cidr')), CHECK (source IN ('operator', 'imported')), CHECK (length(value) > 0) )
